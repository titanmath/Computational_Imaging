<Formule titre="Le résumé en quatre lignes">
$$
\theta = 2\arctan\frac{w_e}{2f}, \qquad
b \approx d + 2{,}44\,\frac{\lambda f}{d}, \qquad
d_\text{opt} \approx \sqrt{2{,}44\,\lambda f}, \qquad
E_\text{écran} \propto \Big(\frac{d}{f}\Big)^2
$$
</Formule>
